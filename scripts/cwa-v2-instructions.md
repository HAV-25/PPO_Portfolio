### 3-2: Call Kimi API

```bash
source .env.local 2>/dev/null || true

ARTICLE_PROMPT=$(cat /tmp/kimi-article-prompt.txt)

node -e "
const https = require('https');
const prompt = require('fs').readFileSync('/tmp/kimi-article-prompt.txt', 'utf8');
const payload = JSON.stringify({
  model: 'kimi-k2.5',
  messages: [{ role: 'user', content: prompt }],
  temperature: 0.3,
  max_tokens: 8192
});
const options = {
  hostname: 'api.moonshot.ai',
  path: '/v1/chat/completions',
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'Authorization': 'Bearer ' + process.env.MOONSHOT_API_KEY
  }
};
const req = https.request(options, res => {
  let data = '';
  res.on('data', chunk => data += chunk);
  res.on('end', () => {
    const parsed = JSON.parse(data);
    if (parsed.error) { console.error('Kimi error:', JSON.stringify(parsed.error)); process.exit(1); }
    require('fs').writeFileSync('/tmp/kimi-response.txt', parsed.choices[0].message.content);
    console.log('Kimi generation complete. Tokens used:', parsed.usage?.total_tokens || 'unknown');
  });
});
req.on('error', e => { console.error('Request error:', e.message); process.exit(1); });
req.write(payload);
req.end();
" 2>&1
```

If the command exits non-zero, log the error and exit — do not attempt to write an article manually.
